{{/* PDF-only link render hook. A link to another local page is dead weight in a static PDF, so we drop the hyperlink and show the target page's title as plain text instead. External URLs, mailto/tel, and same-page anchor links still render as normal clickable tags. */}} {{- $dest := .Destination -}} {{- $isExternal := or (strings.HasPrefix $dest "http://") (strings.HasPrefix $dest "https://") (strings.HasPrefix $dest "mailto:") (strings.HasPrefix $dest "tel:") -}} {{- $isFragment := strings.HasPrefix $dest "#" -}} {{- if or $isExternal $isFragment -}} {{ .Text | safeHTML }} {{- else -}} {{- $path := $dest -}} {{- if strings.Contains $dest "#" -}} {{- $path = index (split $dest "#") 0 -}} {{- end -}} {{- $label := .Text -}} {{- with .Page.GetPage $path -}} {{- $label = .Title -}} {{- end -}} {{ $label }} {{- end -}}
